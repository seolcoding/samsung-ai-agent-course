import os
import time

from langchain_core.messages import HumanMessage, ToolMessage
from langchain_core.tools import tool
from pydantic import BaseModel, Field

import config  # noqa: F401  .env를 읽고 COURSE_ENV에 맞춰 연결 방식을 정한다
from config.check import _mask  # 오류 문구 속 키처럼 보이는 값을 가린다
from config.embeddings import get_embeddings
from config.llm import get_llm
from config.ragaas import post_ragaas


def step(name, fn):
    t = time.time()
    try:
        detail = fn()
        print(f"✔ {name} ({time.time() - t:.1f}초) {detail}")
    except Exception as e:  # noqa: BLE001
        print(f"✖ {name} ({time.time() - t:.1f}초) {type(e).__name__}: {_mask(e)[:200]}")


@tool
def get_weather(city: str) -> str:
    """지정한 도시의 현재 날씨를 조회한다."""
    return "맑음, 기온 22°C"


class Route(BaseModel):
    next: str = Field(description="다음 차례. researcher, writer, FINISH 중 하나")


llm = get_llm()
llm_tools = llm.bind_tools([get_weather])


def tool_call():
    calls = llm_tools.invoke("서울 날씨 알려줘").tool_calls
    assert calls, "tool_calls가 비어 있음(모델이 도구를 고르지 않음)"
    return calls[0]["name"], calls[0]["args"]


def tool_round_trip():
    msgs = [HumanMessage(content="서울 날씨 알려줘")]
    first = llm_tools.invoke(msgs)
    assert first.tool_calls, "tool_calls가 비어 있음"
    msgs += [first, ToolMessage(content="맑음, 기온 22°C", tool_call_id=first.tool_calls[0]["id"])]
    return llm_tools.invoke(msgs).content[:40]


def structured(method=None):
    kw = {"method": method} if method else {}
    r = llm.with_structured_output(Route, **kw).invoke("다음 차례는 writer야. next에 writer를 넣어 줘.")
    assert isinstance(r, Route), f"Route가 아님: {type(r).__name__}"
    return r


def ragaas():
    payload = {
        "index_name": os.getenv("index_name", "mock-index"),
        "permission_groups": ["rag-public"],
        "query_text": "펫보험 보장 내용이 뭐야?",
        "num_result_doc": 3,
        "fields_exclude": ["v_merge_title_content"],
    }
    hits = post_ragaas("retrieve-rrf", payload).json()["hits"]["hits"]
    assert hits, "검색 결과 0건"
    return f"{len(hits)}건, 첫 글 {hits[0]['_source']['merge_title_content'][:30]}"


def langfuse():
    from langfuse import get_client

    assert get_client().auth_check(), "auth_check가 False(키·주소 확인)"
    return os.getenv("LANGFUSE_BASE_URL") or os.getenv("LANGFUSE_HOST")


print("연결 방식:", {k: os.getenv(k) for k in ("LLM_PROVIDER", "EMBED_PROVIDER", "RAG_PROVIDER")})
step("1 기본 호출", lambda: f"{len(llm.invoke('한 단어로만 답하세요: 안녕').content)}자")
step("2 스트리밍", lambda: f"조각 {sum(1 for _ in llm.stream('1부터 5까지 세 줘'))}개")
step("3 도구 고르기 bind_tools", tool_call)
step("4 도구 결과 되돌리기 ToolMessage", tool_round_trip)
step("5 구조화 출력 기본", structured)
step("6 구조화 출력 function_calling", lambda: structured("function_calling"))
step("7 임베딩 1개", lambda: f"{len(get_embeddings().embed_query('테스트'))}차원")
step("8 임베딩 12개(10개씩 나눠 요청)", lambda: f"{len(get_embeddings().embed_documents([f'문장 {i}' for i in range(12)]))}개")
step("9 RAGaaS 검색", ragaas)
step("10 Langfuse 인증", langfuse)

.env.sample에는 이름마다 바로 윗줄에 무엇인지, 요청의 어느 헤더로 가는지, 형식 주의가 영어 주석으로 적혀 있습니다. 컨플루언스 가이드는 값을 줍니다. 둘을 나란히 놓고 이름을 짝지어 값을 붙여 넣으면 됩니다.

5 요청 규격이 다름
원본 이후 사내 규격이 바뀐 경우		헤더 이름이 다름, 컨플루언스에만 있는 필수 헤더, 인증 방식이 다름(예: Authorization: Bearer <티켓>), 코드가 붙이는 경로·본문 옵션이 다름.env로는 못 고칩니다. 값에 Bearer 를 붙이는 식의 우회도 하지 않습니다. 수강생은 config/를 고치지 않습니다. 강사가 course/config/samsung.py·llm.py 한 곳을 고쳐 다 같이 받게 하거나, 그 기능을 건너뜁니다강사

• 오류 문구가 헤더 이름을 말하거나(예: invalid header Authorization, missing header X-…), 위키 예시가 우리 코드와 다른 헤더를 쓰면 5단계입니다. 값을 바꿔 봐도 그대로입니다.

• 헤더는 같고 값만 틀렸다는 문구면 6단계입니다. 값을 다시 복사합니다.

• 알아 둘 것: get_llm()은 OpenAI SDK를 쓰기 때문에 모든 LLM 요청에 Authorization: Bearer samsung-gateway가 자동으로 붙습니다(course/config/llm.py 40번째 줄). 받은 원본 코드도 같았습니다: 가짜 키로 Authorization이 붙은 채 사내 게이트웨이를 불렀습니다(원본 02_openai.py의 OPENAI_API_KEY='api_key'). 원본이 사내에서 돌았다면 게이트웨이는 Authorization을 보지 않는 것으로 보입니다. 그래도 401 문구에 Authorization이 나오면 5단계입니다.

5단계일 때 수업 진행

• 기본 호출이 되고 일부 기능만 막힘: 막힌 셀만 건너뛰고 진행합니다.

• LLM 기본 호출부터 막힘: 1-2·1-3 실습 대부분이 영향을 받습니다. 강의는 그대로 하고, 실습은 코드 읽기·리뷰로 진행하면서 강사 PC에서 미리 실행한 결과 화면을 보여 줍니다. 쉬는 시간에 config를 고치고 --live로 확인한 뒤 다 같이 받게 합니다.

• COURSE_ENV=out(OpenAI) 전환은 개인 키가 있어야 해서 강사가 판단할 일입니다. 고친 내용은 2~4일차에도 영향을 주니 결과표에 남깁니다.

노트북 출력에 [chain/start]·[llm/start]·[chain/end] 같은 로그와 단계별 입력·출력, 실행 시간이 나옵니다. 그림으로 순서를 짚고 → 로그에서 실제 값을 확인하는 순서로 보여 줍니다.

확인이 끝나면 끄기
set_debug(False)

디버그 설정은 같은 커널의 이후 실행에도 적용됩니다. 로그가 너무 길어지면 위 코드를 실행해 끕니다.

8-2. set_debug 더 알기

from langchain_core.globals import set_debug

set_debug(True)    # 켜기: 이 아래 실행부터 단계마다 입력·출력이 찍힘
answer = chain.invoke({"topic": "LLM"})
set_debug(False)   # 끄기: 시연이 끝나면 바로 끈다
  • 보이는 것: [chain/start] [chain:RunnableSequence > prompt:ChatPromptTemplate] Entering …처럼 체인의 각 단계(프롬프트 → 모델 → 파서)가 들어가고 나가는 순서, 입력·출력 값, 걸린 시간(ms).
  • 쓰기 좋은 곳
    • 1-3 실습 2장: prompt | model | parser에서 |가 "앞 단계 출력을 다음 단계 입력으로 넘긴다"는 것을 눈으로 보여 줌
    • 1-3 실습 3장: 모델이 도구 호출 정보를 내는 단계와, 코드가 도구를 실행하는 단계가 따로 찍힘
    • 1-4 RAG 체인: 검색된 문서가 프롬프트 안으로 들어가는 모습
  • 주의
    • 출력이 매우 깁니다. 시연할 셀에서만 켜고 바로 끕니다. 한 번 켜면 커널을 다시 시작할 때까지 모든 셀에 적용됩니다.
    • 프롬프트 내용이 전부 화면에 나옵니다. 키 값은 나오지 않지만, 화면 공유 중이면 내용을 한 번 보고 켭니다.
    • 지금 작업 폴더의 1-3 실습 노트북에는 준비 셀에 set_debug(True)가 들어가 있습니다(직접 고친 것, 커밋 안 됨). 그대로 두면 2장부터 6장 앞까지 계속 켜져 있습니다.

8-4. LangGraph를 한 단계씩 보기

# 노드가 하나 끝날 때마다 그 노드가 바꾼 값만 출력
for event in app.stream(inputs, config, stream_mode="updates"):
    print(event)          # 예: {'chatbot': {'messages': [AIMessage(...)]}}

# 체크포인터가 있을 때: 지금 저장된 상태와 다음에 실행될 노드
state = app.get_state(config)
print(state.next)         # 사람 승인으로 멈췄으면 ('tools',) 같은 값, 끝났으면 ()
state.values["messages"][-1].pretty_print()   # 마지막 메시지를 보기 좋게
  • 쓰기 좋은 곳: 2-1 실습 3장(조건부 엣지로 어느 노드에 갔는지), 4장(도구 노드와 에이전트 노드가 번갈아 도는 모습), 5장(사람 승인에서 멈췄을 때 state.next로 확인). 1-3 실습 6장의 스냅샷도 같은 get_state입니다.
  • stream_mode="updates"는 set_debug보다 훨씬 짧아서 화면 시연에 알맞습니다.
